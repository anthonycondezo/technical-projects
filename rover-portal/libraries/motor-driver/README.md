### Example Makefile
```Makefile
#
# example Makefile - sub-directory /libraries must be present
#

MCU = atmega328p
F_CPU = 16000000

# ---- compilers ----
CC = avr-gcc
AR = avr-ar

# ---- compile flags ----
CFLAGS = -mmcu=$(MCU) -DF_CPU=$(F_CPU)UL -Os -Wall
LDFLAGS = -mmcu=$(MCU)

TARGET = main
SRC = main.c

# ---- library paths ----
USART_LIB = libraries\usart\libusart.a
BUFFER_LIB = libraries\buffer\libbuffer.a
MOTOR_DRIVER_LIB = libraries\motor-driver\libmotor-driver.a
BTN_JOYSTICK_LIB = libraries\btn-joystick\libbtn-joystick.a
LIBS = $(USART_LIB) $(BUFFER_LIB) $(MOTOR_DRIVER_LIB) $(BTN_JOYSTICK_LIB)

PROGAMMER = arduino
PORT = # YOUR PORT NUMBER
BAUD = 115200

all: libs $(TARGET).hex

# ---- library building ----
libs:
	$(MAKE) -C libraries

# ---- application ----
$(TARGET).elf: $(SRC) $(LIBS)
	$(CC) $(CFLAGS) $(LDFLAGS) $^ -o $@

$(TARGET).hex: $(TARGET).elf
	avr-objcopy -O ihex -R .eeprom $< $@

# ---- flashing to microcontroller ----
flash: $(TARGET).hex
	avrdude -c $(PROGAMMER) -p $(MCU) -P $(PORT) -b $(BAUD) -U flash:w:$<

# ---- utilities ----
clean:
	$(MAKE) -C libraries clean
	del *.elf *.hex

```

### Example Code: main.c
Please ensure that the .c file is saved as main.c, otherwise provided makefile will fail to build and
flash to micro-controller.

```c
// main.c
// ATmega328p example
#include "libraries/usart/usart.h"
#include "libraries/buffer/buffer.h"
#include "libraries/motor-driver/motor-driver.h"
#include "libraries/btn-joystick/btn-joystick.h"

#include <stdio.h>
#include <string.h>
#include <arv/io.h>
#include <util/delay.h>
#include <arv/interrupt.h>

#define COMMAND_BUFFER_SIZE 31
#define X_AXIS 0
#define Y_AXIS 1

volatile uint8_t pause = 0;

/** Joystick button will be taken from pin 2 (i.e. pin 10 of arduino) */
void interrupt_init(void) {
    /** Initialise pin2 to be an input and set its voltage to be non-zero (i.e. on) */ 
    DDRB &= ~(1 << DDB2);   // PB2 input
    PORTB |= (1 << PB2);    // pull-up
    /**
    * Enabling external interrrupts (interrupts generated by external pins) 
    * (see p.56 of datasheet - 12.2.4, bit 0 )
    */
    PCICR  |= (1 << PCIE0); // enable PCINT for port B - NOTE: port interrupts enabled by PCICR
    PCMSK0 |= (1 << PCINT2); // NOTE: individual pin interrupts enabled by PCMSK0 
}

/** Button at PORT B pin 2 invokes an interrupt - sets pause  */
ISR(PCINT0_vect) {
    _delay_ms(70);
    if (!(PINB & (1 << PINB2))) {
        // pause printing
        pause = !pause;
        if (pause) {
            uart_println("paused");
        } else {
            uart_println("resumed");
        }
    } 
}

/** 
 * Prints joystick value to puTTY via the USART communication protocol AND
 * maps joystick input axis values to a MOTOR_ACTION
 */
int main(void) {
    int16_t x, y;
    int index = 0;
    char buffer[COMMAND_BUFFER_SIZE];

    adc_init(); // initialise analog-to-digital conversion
    uart_init(); // initiialise USART
    motor_driver_init(); // initialise motor-driver

    /* initialise interrupts */
    interrupt_init();
    sei();

    while (1) {
        if (!pause) {
            x = joystick_axis(adc_read(X_AXIS));
            y = joystick_axis(adc_read(Y_AXIS));
            motor_driver(x, y);
            snprintf(buffer, COMMAND_BUFFER_SIZE, "X %d | Y %d", x, y);
            uart_println(buffer);
            empty_buffer(&index, buffer);
            _delay_ms(500); 
            continue;
        }   
        stop();
    } 
}
```
## Running Example
To run example on your local machine - perform the following terminal commands: 
1) **Update** example makefile with your port number
    ```Makefile
        PORT = # your port
    ```

2) **Build** example application
    ```bash
        make # within project directory 
    ```

3) **Flash** example application
    ```bash
        make flash
    ```

4) **Run** puTTY on your local machine with the following configurations:
- Connection type: serial
- Serial line: your-port-number
- Speed: 9600
